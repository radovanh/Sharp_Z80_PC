  ; Check if SPACE key is pressed (row bit=20h, col bit=5)
  check_space:
      ld   a, 020h         ; strobe row containing SPACE (bit5)
      out  (011h), a
      in   a, (010h)        ; read columns
      bit  5, a             ; test bit 5 (SPACE column)
      ret  z                ; Z set = not pressed
      ; NZ = SPACE is pressed
      ret

  ; Scan all keys: returns bitmask of first pressed key
  scan_all:
      ld   b, 080h         ; start with bit 7
  scan_lp:
      ld   a, b
      out  (011h), a
      in   a, (010h)
      or   a
      ret  nz               ; found a key: A=col bits, B=row bit
      rrc  b
      jr   nc, scan_lp      ; try next row
      xor  a
      ret                   ; no key pressed

BEEP / Sound G850 ONLY

HARDWARE NOTE: The G850 has no internal speaker. Audio output requires a piezo buzzer wired to pins 2 and 7 of the 11-pin serial interface (TTL level, no level converter needed for piezo).

The BASIC BEEP repeat[,level][,length] command generates tones. Frequency formula: 1300000 / (166 + 22 × level) Hz.
